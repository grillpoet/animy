#!/usr/bin/env python3
"""
AnimY 工具：清理脚本产生的备份文件（通配符版）

匹配规则：
  - 文件名中包含 ".bak"（如 foo.kt.bak、foo.kt.bak2、foo.kt.bakP3a）
  - 文件名以 ".old" 结尾

默认 dry-run，加 --delete 才真删。
"""
import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent

SKIP_DIRS = {".git", ".gradle", "build", ".idea", "node_modules", ".kotlin"}

def is_backup(p: Path) -> bool:
    n = p.name
    return (".bak" in n) or n.endswith(".old")

def collect():
    hits = []
    for p in ROOT.rglob("*"):
        if not p.is_file():
            continue
        if any(part in SKIP_DIRS for part in p.parts):
            continue
        if is_backup(p):
            hits.append(p)
    return sorted(hits)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--delete", action="store_true",
                    help="实际删除（默认只列出）")
    args = ap.parse_args()

    hits = collect()

    if not hits:
        print("没有找到备份文件，无需清理。")
        return 0

    total = sum(p.stat().st_size for p in hits)
    print(f"{'删除' if args.delete else '找到'} {len(hits)} 个备份文件，"
          f"共 {total / 1024:.1f} KB：\n")

    for p in hits:
        print(f"  {p.relative_to(ROOT)}  ({p.stat().st_size} B)")

    if not args.delete:
        print()
        print("以上为 dry-run。确认无误后执行：")
        print("  python clean_backups.py --delete")
        return 0

    print()
    failed = []
    for p in hits:
        try:
            p.unlink()
        except Exception as e:
            failed.append((p, e))

    print(f"✓ 已删除 {len(hits) - len(failed)} / {len(hits)} 个文件")
    if failed:
        print()
        for p, e in failed:
            print(f"  ✗ {p}: {e}")
        return 1
    return 0

if __name__ == "__main__":
    sys.exit(main())
